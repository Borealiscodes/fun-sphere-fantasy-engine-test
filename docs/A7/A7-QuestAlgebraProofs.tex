\section*{A7: Quest Algebra Proofs}

This module contains the third set of A7 lemmas, replacing A5 Quest Algebra
axioms with real proofs. These lemmas establish the branching semantics of FSFE
and ensure that QuestStep and TensorCouple interact correctly.

\newpage

% ============================================================
% LEMMA 1 — QUESTSTEP PROGRESS
% ============================================================

\section{Lemma 1: QuestStep Progresses Branch Depth}

\subsection{Lemma Statement}

\begin{verbatim}
Lemma QuestStep_Progress :
  ∀ s : FSFEState,
    QA_depth(QA_next(s.qa)) = QA_depth(s.qa) + 1.
\end{verbatim}

\subsection{Supporting Definitions}

\begin{verbatim}
QA_next : QAState → QAState
QA_depth : QAState → Nat
\end{verbatim}

\subsection{Proof Skeleton}

\begin{verbatim}
Proof.
  intros s.
  unfold QA_next.
  unfold QA_depth.
  (* By definition, QA_next increments depth by 1 *)
  reflexivity.
Qed.
\end{verbatim}

This lemma establishes monotonic branch progression under QuestStep.

\section{Lemma 2: Quest Algebra Stability Under TensorCouple}

\subsection{Lemma Statement}

\begin{verbatim}
Lemma QuestTensor_Stability :
  ∀ s : FSFEState,
    QA_valid(s.qa) →
    QA_valid((TensorCouple(s)).qa).
\end{verbatim}

\subsection{Supporting Definitions}

\begin{verbatim}
TensorCouple(s) := { s with tensor := Tensor_couple(s.tensor) }
QA_valid : QAState → Prop
\end{verbatim}

\subsection{Proof Skeleton}

\begin{verbatim}
Proof.
  intros s Hqa.
  unfold TensorCouple.
  (* TensorCouple does not modify QAState *)
  exact Hqa.
Qed.
\end{verbatim}

This lemma shows that TensorCouple is QA‑neutral.
