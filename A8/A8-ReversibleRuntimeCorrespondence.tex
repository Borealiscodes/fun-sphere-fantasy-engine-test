\section{Lemma 1: Structural Invariant Preservation in Runtime}

\subsection{Lemma Statement}

\begin{verbatim}
Lemma RuntimeInvariant_Preserved :
  ∀ s : FSFEState,
    Inv_struct(s) →
    RuntimeInvariant(StateProject(s)).
\end{verbatim}

\subsection{Supporting Definitions}

\begin{verbatim}
RuntimeInvariant(r) :=
  r.sigma ≥ 0 ∧
  r.lambda ≥ 0 ∧
  r.qa_depth ≥ 0 ∧
  r.tensor_r ≥ 0 ∧
  r.tensor_d ≥ 0 ∧
  r.branch_v = true
\end{verbatim}

\subsection{Proof Skeleton}

\begin{verbatim}
Proof.
  intros s Hinv.
  unfold RuntimeInvariant.
  unfold StateProject.
  destruct Hinv as [Hσ [Hλ [Hqa [Htens Hbranch]]]].
  repeat split; auto.
Qed.
\end{verbatim}

This lemma ensures that structural invariants survive projection into runtime.

\section{Lemma 3: Runtime Reversibility}

\subsection{Lemma Statement}

\begin{verbatim}
Lemma Runtime_Reversible :
  ∀ s : FSFEState,
    StateProject(Λ_inc(Σ_inc(s))) =
    StateProject(Σ_inc(Λ_inc(s))).
\end{verbatim}

\subsection{Proof Skeleton}

\begin{verbatim}
Proof.
  intros s.
  unfold StateProject.
  unfold Σ_inc, Λ_inc.
  (* sigma and lambda increments commute *)
  reflexivity.
Qed.
\end{verbatim}

This lemma ensures that reversible algebraic operators remain reversible at
runtime.
