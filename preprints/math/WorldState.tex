\section*{World Manifold}

\[
W : X_{45} \to \mathcal{S}
\]

\[
\mathcal{S}
= (Z, F, M, E, \mathcal{P}, r, Q, I)
\]

Where:

- \(Z\) = zones  
- \(F\) = factions  
- \(M\) = monster families  
- \(E\) = events  
- \(\mathcal{P}\) = player state  
- \(r\) = rasa tensor  
- \(Q\) = quest algebra  
- \(I\) = inventory tensor  

% ------------------------------------------------------------
% ZONES
% ------------------------------------------------------------

\section*{Zones}

\[
Z = (g_Z, \mathcal{R}_Z, \mathcal{H}_Z, v_k, A_Z)
\]

Zone evolution:

\[
\frac{\partial Z}{\partial t}
= H_Z(Z)
+ \lambda_k Z
+ \nabla_{FS} Z
+ \mathcal{I}_{ZE}(Z, v_k)
+ \mathcal{I}_{ZR}(Z, r)
\]
