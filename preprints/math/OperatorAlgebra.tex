\section*{Reversible Operator Group}

\[
\mathcal{G}_{rev}
= \{ O \in \mathrm{Diff}(X_{45}) \mid O^{-1} \text{ exists} \}
\]

\[
O^{-1} \circ O = \mathrm{id},\qquad
O \circ O^{-1} = \mathrm{id}
\]

% ------------------------------------------------------------

\section*{Lie Algebra of Reversible Operators}

\[
\mathfrak{g}_{rev}
= \{ X \mid \exp(tX) \in \mathcal{G}_{rev} \}
\]

\[
\exp(tX)
= \sum_{n=0}^{\infty} \frac{t^n}{n!} X^n
\]

\section*{Lie Bracket}

\[
[X,Y] = XY - YX

\[
